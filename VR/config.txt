# Black Mesa VR — restart the game after changing RenderScale
# (eye size is picked at CreateDevice).

# Multiplier on the fitted HMD-aspect size. 1.0 is the verified fused path.
#
# RenderScale cannot invent pixels taller than the engine scene buffer.
# The HWND is 2560x1440; the HMD-aspect crop inside it is ~1584x1440, so
# values above 1.0 were fitted back to the window and looked like a no-op.
# L4D2VR uses GetRecommendedRenderTargetSize() as the *scene* size (offscreen),
# not the HWND. BMVR now sizes stereo from `_rt_FullFrameFB` when
# SetRenderTargetFrameBufferSizeOverrides actually grows that RT (crash-sticky
# ff_stereo). If FullFrame stays window-sized, the HMD stays ~1584x1440.
# Do not retry hmd_swap / hmd_native / steamvr_rt SetRT redirect.
# VRRenderScale=1.25
RenderScale=1.0

# Perf: false (default) skips the leftover 16:9 desktop scene render after the
# stereo pair — saves ~30% frame time. Headset flashlight uses gbmatch eyes.
# true = proper 16:9 desktop at extra cost.
DesktopLeftoverRender=false

# Portal 2 VR / L4D2VR right-stick look. Degrees per ms at full deflection.
TurnSpeed=0.25
SnapTurning=false
SnapTurnAngle=45.0

# Fallback viewmodel pos if the per-weapon L4D2VR table has no match.
# Per-weapon pos/ang (including crowbar 19.5,6,-13.5 / -24.5,-6.5,-6) are
# applied in code. These are extra global tweaks on top.
ViewmodelPosOffsetX=0.0
ViewmodelPosOffsetY=0.0
ViewmodelPosOffsetZ=0.0
ViewmodelAngOffsetX=0.0
ViewmodelAngOffsetY=0.0
ViewmodelAngOffsetZ=0.0
# sd805 tilts Vive wands -45°. G2 points more forward.
ControllerPitchTilt=-35.0

# 0.91 was BM/L4D2 scale; user 2026-08-25: ~25% smaller than that → 0.68.
# Crowbar forced to 1.0 in DME. Hands use VrHandsModelScale separately.
ViewmodelScale=0.68

# Compositor timing: true = L4D2VR app PostPresentHandoff (WaitGetPoses off the
# Vulkan queue; ~90fps). false = runtime handoff (waiter WaitGetPoses ~110ms / ~20fps).
CompositorPostPresentHandoff=true

# SteamVR overlay quad (meters, HMD-relative). Smaller/closer than L4D2VR 1.3
# so HUD sits in the central FOV. HudMaxFov does not drive BM HUD.

# Stereo IPD multiplier and extra height (Source units) after 6DOF.

# uses queued (2) via IMaterialSystem::SetThreadMode — not ClientCmd.
# Crash-sticky: bmvr_in_mat_queue.flag (skip-file mat_queue is ignored).
AutoMatQueueMode=true

# Motion / comfort
