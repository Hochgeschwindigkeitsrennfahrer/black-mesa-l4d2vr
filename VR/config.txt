# Black Mesa VR — RenderScale is a multiplier on SteamVR's recommended
# per-eye size. The SteamVR overlay SS / resolution slider applies live
# (eye textures recreate after ~300ms). RenderScale itself is still read
# at DLL load; restart after editing this file.

# Multiplier on SteamVR's recommended per-eye size (OpenVR
# GetRecommendedRenderTargetSize). Independent of the desktop window.
# 16-aligned, cap 4096.
#
# 1.0 = full SteamVR recommended (can be taller than the HWND).
# 0.5–0.75 if GPU bound. Above 1.0 is extra supersampling on top of SteamVR SS.
#
# Dragging SteamVR's resolution / SS slider recreates the eye textures after a
# short settle (no game restart). World G-buffers still need a map load to grow.
# VRRenderScale=1.25
RenderScale=1.0

# Perf: false (default) skips the leftover 16:9 desktop scene render after the
# stereo pair — saves ~30% frame time. Headset flashlight uses gbmatch eyes.
# true = proper 16:9 desktop at extra cost.
DesktopLeftoverRender=false

# Portal 2 VR / L4D2VR right-stick look. Degrees per ms at full deflection.
TurnSpeed=0.25
SnapTurning=false
SnapTurnAngle=45.0

# Fallback viewmodel pos if the per-weapon L4D2VR table has no match.
# Per-weapon pos/ang (including crowbar 19.5,6,-13.5 / -24.5,-6.5,-6) are
# applied in code. These are extra global tweaks on top.
ViewmodelPosOffsetX=0.0
ViewmodelPosOffsetY=0.0
ViewmodelPosOffsetZ=0.0
ViewmodelAngOffsetX=0.0
ViewmodelAngOffsetY=0.0
ViewmodelAngOffsetZ=0.0
# sd805 tilts Vive wands -45°. G2 points more forward.
ControllerPitchTilt=-35.0

# 0.91 was BM/L4D2 scale; user 2026-08-25: ~25% smaller than that → 0.68.
# Crowbar forced to 1.0 in DME. Hands use VrHandsModelScale separately.
ViewmodelScale=0.68

# Compositor timing: true = L4D2VR app PostPresentHandoff (WaitGetPoses off the
# Vulkan queue; ~90fps). false = runtime handoff (waiter WaitGetPoses ~110ms / ~20fps).
CompositorPostPresentHandoff=true

# SteamVR overlay quad (meters, HMD-relative). Smaller/closer than L4D2VR 1.3
# so HUD sits in the central FOV. HudMaxFov does not drive BM HUD.
HudDistance=1.05
HudSize=0.70
HudMaxFov=60
HudDisplayRatio=0.82

# Stereo IPD multiplier and extra height (Source units) after 6DOF.
IPDScale=1.0
HeightOffset=0.0

# Stereo copies blit the shared backbuffer after each RenderView. Queued
# mode 2 (2026-08-26) made that mono, flicker, and ~180fps stutter. Keep 0
# until named per-eye RTs work. Do not ClientCmd mat_queue_mode.
AutoMatQueueMode=false

# L4D2VR VR-menu MSAA on the private eye RTs (not mat_antialias). 0 / 2 / 4 / 8 / 16.
# Restart required. OpenVR submits a non-MSAA resolve copy. This is GPU cost,
# not a performance win. On the HMD-fb blit path the world is already resolved
# into the window, so geometry edges stay aliased; gloves drawn into the eyes
# still get MSAA. Named per-eye RTs would AA the scene like L4D2VR.
AntiAliasing=0

# Motion / comfort
Haptics=true
HideCrosshair=true
# MatchHmdHz no longer caps fps_max at the headset rate (that was 90 on G2).
# fps_max is 0 (uncapped); SteamVR WaitGetPoses still paces compositor submit.
MatchHmdHz=false
DisableViewBob=true
LeftHanded=false
RecenterResetsYaw=true
HideLocalPlayerModel=true

# Hybrid VR hands (not L4D2VR NativeViewmodelHandsOnly). Hide HEV/marine FP
# arms on the native v_* viewmodel; SteamVR vr_glove GLB at each controller.
# Debug boxes only if gloves fail (or set true to overlay). NativeViewmodelHandsOnly stays off.
HideViewmodelArms=true
VrHandsGlovesEnabled=true
VrHandsModelScale=0.85
# SteamVR glove local Rx,Ry,Rz. Yaw 180 points fingers with the controller.
VrHandsPoseRotationOffset=0,180,0
# Negative Y lowers both hands in view (separate from weapon oz). Controller
# tracking position stays; only the GLB mesh offset changes.
VrHandsPoseOffsetMeters=0,-0.035,0
VrHandsUseHevGloves=true
VrHandsDebugBoxes=false
VrHandHud=true
